7. 完整流程测试（Python 内联）
# 测试 sim_1 ~ sim_4 串联
python3 -c "
import sim_1
import sim_2
import sim_3
import sim_4

seq = sim_1.fill_placeholder_positions('ACxxxxxxCxxxxxxCG', fill_all=True)
print('Sequence:', seq)

pdb = sim_2.generate_conformation(seq, verbose=False)
print('PDB:', pdb)

pdbqt = sim_3.pdb_to_pdbqt(pdb, verbose=False)
print('PDBQT:', pdbqt)

result = sim_4.run_vina_docking(pdbqt, target_name='1LYZ', exhaustiveness=1, num_modes=1, verbose=False)
best = result.get_best()
print('Energy:', best.binding_energy if best else 'None')
"
